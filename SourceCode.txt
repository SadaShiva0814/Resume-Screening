def parse_resume(text, html_header=None):
    if not text or len(text.strip()) < 20:
        return _empty_result()
    
    nlp = _get_nlp()
    Strategy: Hierarchical extraction (Contact -> Sections -> Skills -> Experience)
    contact = _extract_contact_info(text, nlp, html_header=html_header)
    
    Segment into sections using precompiled regex patterns
    sections = _segment_sections(text)
    
    SEMANTIC FALLBACK: If header detection fails, use NLP to classify paragraphs
    if _is_poorly_segmented(sections):
        sections = _contextual_segment(text, nlp)
    
    Structured entity extraction from parsed sections
    extracted_skills = _extract_skills(sections.get('skills', ''), text, nlp)
    years_exp = _estimate_years_experience(sections.get('experience', ''), text, sections)
    edu_level = _detect_education_level(sections.get('education', ''), text)
    
    return {
        'contact': contact,
        'sections': sections,              # Raw text for ML scoring
        'extracted_skills': extracted_skills,
        'years_of_experience': years_exp,
        'education_level': edu_level,
    }

3. compute_section_scores(parsed_resume, jd_embedding, jd_text="")
Implements Signal Isolation. This prevents irrelevant text from diluting strong matches.

def compute_section_scores(parsed_resume, jd_embedding, jd_text=""):
    sections = parsed_resume.get('sections', {})
    jd_skills = _extract_jd_requirements(jd_text) # Target skills from JD
    section_scores = {}
    
    for section_name, section_text in sections.items():
        Split into paragraphs to isolate "signals"
        paragraphs = [p for p in section_text.split('\n\n') if len(p.strip()) > 20]
        
        para_scores = []
        for p in paragraphs:
            p_vec = encode_text(p)
            similarity = cosine_similarity(p_vec, jd_embedding)
            
            SIGNAL DENSITY: Boost paragraphs that concentrate specific JD keywords
            p_matches = sum(1 for s in jd_skills if s.lower() in p.lower())
            density_factor = 0.8 + min(0.6, (p_matches / len(jd_skills)) * 2.0)
            
            para_scores.append(similarity * density_factor)
        
        SIGNAL ISOLATION: The section score is determined by the BEST matching paragraph
        score = max(para_scores) if para_scores else 0.0
        
        Ontological Bonus: Higher credits for direct matches in the Skill section
        if section_name == 'skills':
            score += 0.10 + min(0.35, (matches / len(jd_skills)) * 0.55)
            
        section_scores[section_name] = max(0.0, min(1.3, score)) 
    
    return {'section_scores': section_scores}

1. VectorStore.build_index(embeddings_matrix, ids=None)
Encapsulates FAISS logic for fast semantic retrieval.

def build_index(self, embeddings_matrix, ids=None):
    FAISS IndexFlatIP: Uses Inner Product for Similarity
    Optimized in C++ for vector-matrix multiplication
    index = faiss.IndexFlatIP(self.vector_dim)
    
    Ensure strict float32 and L2 normalization for Cosine Similarity equivalence
    embeddings_matrix = np.array(embeddings_matrix, dtype=np.float32)
    faiss.normalize_L2(embeddings_matrix)
    
    index.add(embeddings_matrix)
    id_mapping = {i: _id for i, _id in enumerate(ids)} if ids else {}
    
    return index, id_mapping

def search(self, index, query_vector, k=10):
    k = min(k, index.ntotal)
    Prepare query_vector (L2 normalization)
    query_vector = np.array(query_vector, dtype=np.float32).reshape(1, -1)
    faiss.normalize_L2(query_vector)
    
    Perform search on built index
    distances, indices = index.search(query_vector, k)
    return distances[0], indices[0]

2. _compute_adaptive_weights(shortlisted, rejected)
Calculates how much a section differentiates successful candidates from failures.

def _compute_adaptive_weights(shortlisted, rejected):
    Differentiating Logic:
    # discriminant = mean_score(shortlisted_section) - mean_score(rejected_section)
    sections = ['skills', 'experience', 'education', 'projects', 'summary']
    
    discriminant = {}
    for section in sections:
        s_mean = np.mean([f['section_scores'].get(section, 0) for f in shortlisted])
        r_mean = np.mean([f['section_scores'].get(section, 0) for f in rejected])
        discriminant[section] = s_mean - r_mean
    
    Shift base weights based on learning_rate
    adjusted = {}
    for section in sections:
        base = Config.DEFAULT_WEIGHTS.get(section, 0.2)
        adjustment = Config.LEARNING_RATE * discriminant[section]
        adjusted[section] = max(0.02, base + adjustment)
    
    Normalize to sum to 1.0 (Sum-to-100 logic)
    total = sum(adjusted.values())
    return {k: round(v / total, 4) for k, v in adjusted.items()}

def screen():
    1. Capture user inputs and job category
    job_description = request.form.get('job_description', '').strip()
    job_category = request.form.get('job_category', 'general').lower()
    files = request.files.getlist('files[]')
    
    2. Initialize a persistent Screening Session
    session_id = create_session(job_description, job_category, num_resumes=len(files))
    
    3. Offload high-latency AI tasks to a background thread
    def _run_pipeline(app, sid, inputs, jd, cat):
        with app.app_context():
            The orchestrator is called in background
            candidates = screen_resumes(inputs, jd, job_category=cat)
            
            Save results and update status once complete
            save_candidates(sid, candidates)
            update_session_status(sid, 'completed')
            
    Launch pipeline without blocking the user
    threading.Thread(target=_run_pipeline, args=(current_app, session_id, ...)).start()
    
    return jsonify({'success': True, 'session_id': session_id, 'redirect': f'/status/{session_id}'})
